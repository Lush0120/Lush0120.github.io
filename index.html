Dashboards en Power BI y proyectos de análisis de datos de mis cursos.

 

SELECT c.nombre                          AS candidato,
       p.nombre                          AS prueba,
       i.nota,
       CASE
         WHEN i.nota >= 4.5 THEN 'Avanzado'
         WHEN i.nota >= 3.5 THEN 'Intermedio'
         ELSE 'Básico'
       END                               AS nivel,
       (SELECT ROUND(100.0 * SUM(CASE WHEN r.correcta THEN 1 ELSE 0 END)
                     / COUNT(*), 1)
          FROM respuestas r
         WHERE r.intento_id = i.id)      AS porcentaje_aprobacion
  FROM intentos i
  JOIN candidatos c ON c.id = i.candidato_id
  JOIN pruebas    p ON p.id = i.prueba_id
 ORDER BY c.nombre, p.nombre;

-- candidato     | prueba        | nota | nivel      | porcentaje_aprobacion
-- Ana Pérez     | Python básico |  4.6 | Avanzado   |  92.0
-- Luis Gómez    | SQL           |  3.8 | Intermedio |  76.0
